cmake_minimum_required(VERSION 3.24)
project(DX12Engine VERSION 0.1.0 LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "DX12Engine requires Windows x64 and MSVC. Use the Visual Studio 2022 x64 preset.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Supported configurations" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set_property(GLOBAL PROPERTY USE_FOLDERS ON)
option(ENGINE_BUILD_EDITOR "Build the editor" ON)
option(ENGINE_BUILD_GAME "Build the game host" ON)
option(ENGINE_ENABLE_HARNESS "Build bounded runs and diagnostic ledgers" ON)
option(ENGINE_ENABLE_CONTROL_CHANNEL "Build the opt-in local control channel" ON)
option(ENGINE_ENABLE_TRACY "Build Tracy CPU/GPU instrumentation" OFF)
option(ENGINE_ENABLE_NAV "Link navigation support" ON)
option(ENGINE_ENABLE_MEDIA "Link FFmpeg media decoding" OFF)
option(ENGINE_ENABLE_VIDEO "Link video playback" OFF)
option(ENGINE_ENABLE_FACE_TRACKING "Link face tracking (requires Media and ONNX Runtime)" OFF)
option(ENGINE_ENABLE_STREAMLINE "Link NVIDIA Streamline (requires its separately acquired SDK)" OFF)
set(ENGINE_FFX_SDK_DIR "${PROJECT_SOURCE_DIR}/DX12/external/FidelityFX" CACHE PATH "FidelityFX SDK root")
set(ENGINE_NRD_SDK_DIR "${PROJECT_SOURCE_DIR}/DX12/external/NRD" CACHE PATH "NRD SDK root")
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
include(CTest)
add_subdirectory(DX12)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_NAME DX12Engine)
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_FILE_NAME "DX12Engine-${PROJECT_VERSION}-win64")
set(CPACK_PACKAGE_DIRECTORY "${PROJECT_SOURCE_DIR}/DX12/build/package")
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY ON)
include(CPack)
